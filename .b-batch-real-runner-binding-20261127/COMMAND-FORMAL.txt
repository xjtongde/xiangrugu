# ── 步骤 0：从固定运行时提交导出到干净目录、生成 PIN_COMMIT、安装模块清单 ──
# ⚠ 运行位置（重要）：
#   * git archive / rev-parse 需在**有该提交的 git clone** 上执行——主机 32 与
#     xiangrugu-dev 容器内均无可用仓库（容器内无 git，32 上 deploy 非 git 仓库）。
#     因此步骤 0 的前段在交付方的 clone 上执行，产出 tar 包。
#   * 解包与后续步骤在 xiangrugu-dev 容器内执行；目标目录必须位于容器可写挂载点
#     /data/var（容器根文件系统只读，/workspace 只读）。
#   * 清单 .b-batch-final-pin-20261127/manifest/PIN_MODULES.json 随交付件一同送入。
# ── 变量必须先于任何使用定义（本盒修正）──
# 原先步骤 0 有 8 处硬编码 /data/var/b-batch/code，而 $CODE_ROOT 到下面才 export，
# 导致换目录必须逐处手工替换、且漏改一处不会报错只会跑在旧目录。现在全部走变量。
PIN=a5d12b13ad87b74fea5e8e7b2fb8a2c901bbad9e
export CODE_ROOT=/data/var/b-batch/code/xrg-batch
export WORK=/data/var/b-batch/work
export PAYLOAD=/data/var/b-batch/payload.bin
BOX=/data/var/b-batch

# (0a) 在交付方 clone 上：从固定提交导出为 tar 包
git archive --format=tar --prefix=xrg-batch/ "$PIN" | gzip > /tmp/xrg-batch-$PIN.tgz
git rev-parse "$PIN" > /tmp/xrg-batch-PIN.txt

# (0b) 送入容器并解包到干净目录
docker exec -i xiangrugu-dev sh -c \
  "rm -rf \"$BOX/code\" && mkdir -p \"$BOX/code\" \
   && tar xzf - -C \"$BOX/code\"" < /tmp/xrg-batch-$PIN.tgz
docker exec -i xiangrugu-dev sh -c \
  "cat > \"$CODE_ROOT/PIN_COMMIT\"" < /tmp/xrg-batch-PIN.txt

# 安装模块清单到 --code-root。清单不入 git 树，是独立交付件；
# 必须装上，否则 runner 的 code_pin 校验与预检的 manifest_covers_* 失败。
docker cp .b-batch-final-pin-20261127/manifest/PIN_MODULES.json \
  xiangrugu-dev:"$CODE_ROOT/PIN_MODULES.json"

# 核对安装结果（四项都必须成立）。
# ⚠ 三条约定（本盒按 a04 复核意见补齐）：
#   1. 执行命令的 docker exec **不带 -i 且显式 </dev/null**——与下方步骤 1/2 同规则；
#      -i 只在 (0b) 用 tar/cat 经 stdin 传数据时才必需。
#   2. PIN_COMMIT 比对用 cat 单独取值再比，不用 test "$X" = $PIN 嵌套引号。
#   3. 清单必须是**常规文件**：docker cp /dev/stdin 会写出指向 /proc/self/fd/0 的
#      坏符号链接且返回 0，所以除 -f 还要验 ! -L。
echo "--- 0c-1 PIN_COMMIT ---"
docker exec -w "$BOX" xiangrugu-dev sh -c \
  'cat "$1/PIN_COMMIT"' sh "$CODE_ROOT" </dev/null > /tmp/xrg-pin-readback.out
RC=$?
PIN_READBACK=$(cat /tmp/xrg-pin-readback.out)
if [ "$RC" -ne 0 ] || [ "$PIN_READBACK" != "$PIN" ]; then
  echo "PIN_COMMIT_MISMATCH rc=$RC readback=$PIN_READBACK want=$PIN"; exit 1
fi
echo "PIN_COMMIT_OK=$PIN_READBACK"

echo "--- 0c-2/3 manifest regular file + 16 entries ---"
docker exec -w "$BOX" xiangrugu-dev sh -c \
  'test -f "$1/PIN_MODULES.json" && ! test -L "$1/PIN_MODULES.json" && echo REGULAR_FILE_OK' \
  sh "$CODE_ROOT" </dev/null
RC=$?
[ "$RC" -eq 0 ] || { echo "MANIFEST_NOT_REGULAR_FILE rc=$RC"; exit 1; }
docker exec -w "$BOX" -e PYTHONPATH="$CODE_ROOT/datamgmt/src" xiangrugu-dev python3 -c \
  "import json,os;m=json.load(open(os.environ['CODE_ROOT']+'/PIN_MODULES.json'));\
   assert len(m)==16,len(m);print('PIN_MODULES entries =',len(m))" </dev/null
RC=$?
[ "$RC" -eq 0 ] || { echo "MANIFEST_NOT_16_ENTRIES rc=$RC"; exit 1; }

# 从固定导出树启动 runner：-m 用的 xiangrugu_datamgmt 必须解析到该导出树，
# 不能落到已安装的 site-packages 版本。
# 同样不用管道取退出码：先落盘再判，避免把 grep/tail 的状态当成被测结论。
docker exec -w "$BOX" -e PYTHONPATH="$CODE_ROOT/datamgmt/src" \
  xiangrugu-dev python3 -c "import xiangrugu_datamgmt as m;print(m.__file__)" \
  </dev/null > /tmp/xrg-step0-probe.out
RC=$?
if [ "$RC" -ne 0 ] || ! grep -q "^$CODE_ROOT/datamgmt/src/" /tmp/xrg-step0-probe.out; then
  echo "RUNTIME_NOT_FROM_EXPORT_TREE rc=$RC observed=$(cat /tmp/xrg-step0-probe.out)"
  exit 1
fi
echo "STEP0_PROBE_EXIT=$RC module=$(cat /tmp/xrg-step0-probe.out)"

# ─────────────────────────────────────────────────────────────────────
# 执行位置总则
#   位置 A = 交付方 clone（有该提交的 git 仓库）
#   位置 B = 主机 32（docker 与卷所在；预检/装载在 xiangrugu-dev 容器内跑）
#   固定 Gitea 工件（清单、命令、scope/expected、授权文本、合同）由交付方 clone
#   经 `docker cp` 或 tar-over-ssh 送入容器内的 /data/var/b-batch/work/，
#   步骤 1/2 一律在该工作目录下用相对路径引用。
# ─────────────────────────────────────────────────────────────────────
export PIN=a5d12b13ad87b74fea5e8e7b2fb8a2c901bbad9e
# CODE_ROOT / WORK / PAYLOAD 已在文件顶部定义，此处不再重复（避免两处不一致）。

# ═══════════════════════ 位置 B：步骤 1 只读预检 ═══════════════════════
# ⚠ 执行边界（a04 实测纠正，本盒修复）：
#   `$WORK`=/data/var/...、`/run/secrets/...` 全是**容器内**路径；主机 32 上
#   既无 /data/var/b-batch 也无 /run/secrets，且宿主机 python3 无法
#   import xiangrugu_datamgmt（ModuleNotFoundError）。上一版第 73/97 行是裸
#   `python3`，在宿主机执行，按原文根本跑不起来。
#   现在被测进程一律由**显式 docker exec** 在容器内启动：
#     -w "$WORK"                      给实际运行进程设定工作目录
#     -e PYTHONPATH="$CODE_ROOT/..."  给实际运行进程（不只是给探针）
#   上一版只把 PYTHONPATH 给了探针，被测进程没拿到。
#
# ⚠ 命令调用**不加 -i**，并显式 </dev/null（本盒实测发现）：
#   `docker exec -i` 会把容器的 stdin 接到脚本自己的 stdin。当整份命令经
#   heredoc / `bash -s` / 管道喂入时，-i 会吞掉脚本尚未读取的正文——
#   表现为第一条探针就永久挂起，且没有任何报错。只有 (0b) 那种「用 tar/cat
#   通过 stdin 传数据」的步骤才需要 -i；执行命令一律不带 -i 且 </dev/null。
#
# ⚠ 退出码必须短路（a04 实测纠正）：上一版只有 `echo "PREFLIGHT_EXIT=$?"`，
#   注释写着「退出码 0 才继续」但没有实现——预检失败后仍进入装载段，
#   且脚本自身恒以 0 结束，把失败伪装成成功。
#   注意取退出码的方式：RC=$? 紧跟被测命令之后，**不用管道**
#   （`cmd | tail` 会把 tail 的状态当结论）。
#
# 工作目录由 docker exec -w 在容器内设定；下方输入改用相对 $WORK 的路径，
# 这样"进程的实际 cwd"与"路径的解析基准"是同一件事，不再靠字面绝对路径掩盖。
docker exec -w "$WORK" -e PYTHONPATH="$CODE_ROOT/datamgmt/src" xiangrugu-dev \
  python3 -c "import xiangrugu_datamgmt as m;print(m.__file__)" \
  </dev/null > /tmp/xrg-runtime-probe.out
RC=$?
if [ "$RC" -ne 0 ] || ! grep -q "^$CODE_ROOT/datamgmt/src/" /tmp/xrg-runtime-probe.out; then
  echo "RUNTIME_NOT_FROM_EXPORT_TREE rc=$RC observed=$(cat /tmp/xrg-runtime-probe.out)"
  exit 1
fi

docker exec -w "$WORK" -e PYTHONPATH="$CODE_ROOT/datamgmt/src" xiangrugu-dev \
  python3 .b-batch-real-runner-binding-20261127/preflight.py \
  --phase phase2-production \
  --scope       ".b-batch-real-runner-binding-20261127/formal-scope.json" \
  --expected    ".b-batch-real-runner-binding-20261127/formal-expected.json" \
  --approval-text    ".b-batch-real-load-20261127/approval-text-verbatim.txt" \
  --approval-table   "datamgmt/contracts/releases/2026-10-06-bootstrap/b-approval-table-v2.json" \
  --payload     "$PAYLOAD" \
  --code-root   "$CODE_ROOT" \
  --code-commit "$PIN" \
  --evidence-out /data/var/b-batch/preflight-formal.json </dev/null
RC=$?
echo "PREFLIGHT_EXIT=$RC"
# 短路：非 0 立即以同一退出码结束，禁止进入装载段。2 = 检查失败 / 4 = 输入缺失。
[ "$RC" -eq 0 ] || { echo "ABORT_BEFORE_LOAD: preflight rc=$RC"; exit "$RC"; }

# ─────────────────────────────────────────────────────────────────────
# ⚠ 替身标记（LOAD_ENTRY_CALLED）**必须在容器内判读**，不得在宿主机 test -f。
#   a04 实测：HOST_SEES_MARKER=NO / CONTAINER_SEES_MARKER=YES ——
#   宿主机看不到容器内的文件。若在宿主侧判读，"标记不存在"永远是真，
#   负对照会退化成**假阴性**（无论装载入口有没有被调用都报通过）。
#   正确形态：
#     docker exec xiangrugu-dev sh -c 'test -f "$MARK" && echo YES || echo NO'
# ─────────────────────────────────────────────────────────────────────

# ═══════════════════════ 位置 B：步骤 2 正式装载 ═══════════════════════
# 同上：runner 也必须在容器内执行，PYTHONPATH 与 -w 给**实际运行进程**。
# XNG_LOAD_PASSWORD 由宿主机读取 secret 后经 -e 传入容器环境，
# 不出现在命令行上（避免进入 ps / shell 历史）。
# ⚠ 放行前必须显式确认该 secret 在**宿主机 32** 上存在且可读（本盒按边界要求未 cat、
#   未验证其存在性；a04 同样未验证——所以这一条现在是"待确认"而非"已成立"）。
#   取码不用管道：先单独取值再比，避免 test "$X" = $PIN 嵌套引号失效。
docker exec xiangrugu-dev true   # 容器可用性前置检查
test -r /run/secrets/xrg_loader_password \
  || { echo "SECRET_NOT_PRESENT_ON_HOST_32: /run/secrets/xrg_loader_password"; exit 1; }
export XNG_LOAD_PASSWORD="$(cat /run/secrets/xrg_loader_password)"
docker exec -w "$WORK" \
  -e PYTHONPATH="$CODE_ROOT/datamgmt/src" \
  -e XNG_LOAD_PASSWORD xiangrugu-dev \
  python3 -m xiangrugu_datamgmt.cbdb_social_addr_runner \
  --phase phase2-production \
  --code-root   "$CODE_ROOT" \
  --code-commit "$PIN" \
  --contract    "datamgmt/contracts/releases/2026-10-06-bootstrap/b-single-table-frozen-candidate.json" \
  --approval-table "datamgmt/contracts/releases/2026-10-06-bootstrap/b-approval-table-v2.json" \
  --approval-text  ".b-batch-real-load-20261127/approval-text-verbatim.txt" \
  --approval-text-sha256 4d250c8e08ea8136e29e35aa436812f11b5eec59a100847f130c11f39b7e2946 \
  --scope       ".b-batch-real-runner-binding-20261127/formal-scope.json" \
  --expected    ".b-batch-real-runner-binding-20261127/formal-expected.json" \
  --payload     "$PAYLOAD" \
  --scratch     /data/var/b-batch \
  --evidence-out /data/var/b-batch/evidence-formal.json </dev/null
RC=$?
echo "LOAD_EXIT=$RC"
# runner 的退出码原样传给调用者：不再用 `echo` 收尾把脚本自身拉回 0。
exit "$RC"
