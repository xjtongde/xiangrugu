# ── 步骤 0：从固定运行时提交导出到干净目录、生成 PIN_COMMIT、安装模块清单 ──
# ⚠ 运行位置（重要）：
#   * git archive / rev-parse 需在**有该提交的 git clone** 上执行——主机 32 与
#     xiangrugu-dev 容器内均无可用仓库（容器内无 git，32 上 deploy 非 git 仓库）。
#     因此步骤 0 的前段在交付方的 clone 上执行，产出 tar 包。
#   * 解包与后续步骤在 xiangrugu-dev 容器内执行；目标目录必须位于容器可写挂载点
#     /data/var（容器根文件系统只读，/workspace 只读）。

PIN=a5d12b13ad87b74fea5e8e7b2fb8a2c901bbad9e

# (0a) 在交付方 clone 上：从固定提交导出为 tar 包
git archive --format=tar --prefix=xrg-batch/ "$PIN" | gzip > /tmp/xrg-batch-$PIN.tgz
git rev-parse "$PIN" > /tmp/xrg-batch-PIN.txt

# (0b) 送入容器并解包到干净目录
docker exec -i xiangrugu-dev sh -c \
  "rm -rf /data/var/b-batch/code && mkdir -p /data/var/b-batch/code \
   && tar xzf - -C /data/var/b-batch/code" < /tmp/xrg-batch-$PIN.tgz
docker exec -i xiangrugu-dev sh -c \
  "cat > /data/var/b-batch/code/xrg-batch/PIN_COMMIT" < /tmp/xrg-batch-PIN.txt

# 安装模块清单到 --code-root。清单不入 git 树，是独立交付件；
# 必须装上，否则 runner 的 code_pin 校验与预检的 manifest_covers_* 失败。
docker cp .b-batch-final-pin-20261127/manifest/PIN_MODULES.json \
  xiangrugu-dev:/data/var/b-batch/code/xrg-batch/PIN_MODULES.json

# 核对安装结果（三项都必须成立）
docker exec xiangrugu-dev sh -c \
  'test "$(cat /data/var/b-batch/code/xrg-batch/PIN_COMMIT)" = \
       "a5d12b13ad87b74fea5e8e7b2fb8a2c901bbad9e"'
docker exec xiangrugu-dev sh -c 'test -f /data/var/b-batch/code/xrg-batch/PIN_MODULES.json'
docker exec xiangrugu-dev python3 -c \
  "import json;m=json.load(open('/data/var/b-batch/code/xrg-batch/PIN_MODULES.json'));\
   assert len(m)==16,len(m);print('PIN_MODULES entries =',len(m))"

  | grep -q "^/data/var/b-batch/code/xrg-batch/datamgmt/src/" \
